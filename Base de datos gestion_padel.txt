CREATE DATABASE gestion_padel;
USE gestion_padel;

CREATE TABLE Usuario (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    rol ENUM('admin', 'user') NOT NULL
);


CREATE TABLE Canchas (
    id_cancha INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    tipo VARCHAR(20),
    precio_hora DECIMAL(5,2),
    esta_disponible BOOLEAN DEFAULT TRUE
);


CREATE TABLE Reservas (
    id_reserva INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT,
    id_cancha INT,
    fecha DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    total_pago DECIMAL(5,2),
    metodo_pago ENUM('Tarjeta', 'Efectivo', 'Bono') not null,
    CONSTRAINT fk_reserva_usuario FOREIGN KEY (id_usuario) REFERENCES Usuario(id_usuario),
    CONSTRAINT fk_reserva_cancha FOREIGN KEY (id_cancha) REFERENCES Canchas(id_cancha)
);

SELECT nombre, email 
FROM Usuario 
WHERE rol = 'user';

SELECT nombre, precio_hora 
FROM Canchas 
WHERE tipo = 'Cristal' AND esta_disponible = TRUE;

SELECT * FROM Reservas 
WHERE fecha = '2024-05-01';

SELECT metodo_pago, SUM(total_pago) AS ingresos_totales
FROM Reservas 
GROUP BY metodo_pago;

SELECT fecha, hora_inicio, total_pago 
FROM Reservas 
WHERE id_usuario = 2 
ORDER BY fecha DESC;

SELECT c.nombre AS pista, r.fecha, r.hora_inicio
FROM Reservas r
JOIN Canchas c ON r.id_cancha = c.id_cancha;

SELECT u.nombre, COUNT(r.id_reserva) AS numero_reservas
FROM Usuario u
JOIN Reservas r ON u.id_usuario = r.id_usuario
GROUP BY u.id_usuario
ORDER BY numero_reservas DESC
LIMIT 5;

SELECT c.tipo, SUM(r.total_pago) AS total_recaudado
FROM Canchas c
JOIN Reservas r ON c.id_cancha = r.id_cancha
GROUP BY c.tipo;

SELECT * FROM Reservas 
WHERE hora_inicio >= '18:00:00';

SELECT AVG(total_pago) AS precio_medio_bono
FROM Reservas 
WHERE metodo_pago = 'Bono';
